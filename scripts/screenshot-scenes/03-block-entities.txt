# Block entities: the block types whose 26.2 model file is *empty*, so every
# one of them is a hole in the terrain mesh until a render source supplies it.
#
# Banners (layered loom patterns composited into one flag mesh, with sway),
# decorated pots (four independent sherd faces), chests and shulker boxes,
# a conduit (shell + spinning eye), a lit campfire cooking real items, a
# lectern holding a book, a bell, an enchanting table (four animated book
# values, none of them on the wire), and a beacon firing its beam.

@size 768 432
@camera 0.0 66.3 4.6
@look 0.0 64.9 12.4
@fov 68
@wait 3000

fill -16 63 0 16 90 32 minecraft:air
kill @e[type=!player,x=-14,y=60,z=2,dx=28,dy=30,dz=28]
fill -14 63 2 14 63 30 minecraft:smooth_stone
fill -9 63 6 9 63 20 minecraft:polished_andesite
fill -9 63 20 9 63 20 minecraft:chiseled_polished_blackstone

# --- beacon, centre and furthest back -------------------------------------
fill -1 63 17 1 63 19 minecraft:iron_block
setblock 0 64 18 minecraft:beacon
setblock -3 64 18 minecraft:sea_lantern
setblock 3 64 18 minecraft:sea_lantern

# --- banners, flanking the beacon -----------------------------------------
setblock -6 64 17 minecraft:red_banner[rotation=8]{patterns:[{pattern:"minecraft:stripe_bottom",color:"white"},{pattern:"minecraft:cross",color:"black"},{pattern:"minecraft:border",color:"yellow"}]}
setblock -5 64 17 minecraft:blue_banner[rotation=8]{patterns:[{pattern:"minecraft:rhombus",color:"white"},{pattern:"minecraft:circle",color:"light_blue"},{pattern:"minecraft:curly_border",color:"white"}]}
setblock -4 64 17 minecraft:white_banner[rotation=8]{patterns:[{pattern:"minecraft:gradient_up",color:"purple"},{pattern:"minecraft:creeper",color:"lime"},{pattern:"minecraft:bordure_indented",color:"black"}]}
setblock 4 64 17 minecraft:green_banner[rotation=8]{patterns:[{pattern:"minecraft:half_horizontal",color:"lime"},{pattern:"minecraft:skull",color:"white"},{pattern:"minecraft:border",color:"black"}]}
setblock 5 64 17 minecraft:orange_banner[rotation=8]{patterns:[{pattern:"minecraft:diagonal_up_right",color:"magenta"},{pattern:"minecraft:flower",color:"yellow"},{pattern:"minecraft:triangles_bottom",color:"black"}]}
setblock 6 64 17 minecraft:black_banner[rotation=8]{patterns:[{pattern:"minecraft:mojang",color:"white"},{pattern:"minecraft:border",color:"red"}]}

# --- middle row: chests, lectern, campfire, bell ---------------------------
setblock -6 64 13 minecraft:chest[facing=south,type=left]
setblock -5 64 13 minecraft:chest[facing=south,type=right]
setblock -3 64 13 minecraft:ender_chest[facing=south]
setblock -1 64 13 minecraft:enchanting_table
setblock 1 64 13 minecraft:lectern[facing=south,has_book=true]{Book:{id:"minecraft:written_book",count:1,components:{"minecraft:written_book_content":{title:"Lodestone",author:"lodestone",pages:[{raw:"hello"}]}}},Page:0}
setblock 3 64 13 minecraft:campfire[lit=true,facing=south]{Items:[{Slot:0b,id:"minecraft:porkchop",count:1},{Slot:1b,id:"minecraft:potato",count:1}]}
setblock 5 64 13 minecraft:bell[facing=south,attachment=floor]
setblock 6 64 13 minecraft:trapped_chest[facing=south]

# --- front row: shulkers, pots, conduit ------------------------------------
setblock -5 64 9 minecraft:purple_shulker_box[facing=up]
setblock -4 64 9 minecraft:light_blue_shulker_box[facing=south]
setblock -2 64 9 minecraft:decorated_pot[facing=south]{sherds:["minecraft:angler_pottery_sherd","minecraft:brewer_pottery_sherd","minecraft:danger_pottery_sherd","minecraft:explorer_pottery_sherd"]}
setblock 0 64 9 minecraft:decorated_pot[facing=south]{sherds:["minecraft:heart_pottery_sherd","minecraft:friend_pottery_sherd","minecraft:mourner_pottery_sherd","minecraft:skull_pottery_sherd"]}
setblock 2 64 9 minecraft:barrel[facing=south]
setblock 4 64 9 minecraft:lime_shulker_box[facing=up]
setblock 5 64 9 minecraft:blue_shulker_box[facing=south]

# Three block-entity types are deliberately NOT in this scene, because they
# render wrongly today and a screenshot must not launder a defect:
#
#   * a **conduit**'s shell draws as a huge translucent blue sheet over the
#     whole stage, not a 1-block cage;
#   * a **skull/head** draws as a plain untextured cube — the face texture
#     never reaches its front quad (measured on skeleton_skull and
#     zombie_head, camera square on to `rotation=8`);
#   * a **dragon_head** draws nothing at all, while the skulls beside it draw.
#
# All three are block-entity-rendered types, so they belong here on merit.
# Put them back when they are fixed.

tp Lodestone 0.0 64.0 28.0 180 8
